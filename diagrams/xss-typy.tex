% (dashed, open heads), self messages. Red marks the messages that carry the malicious string.
\begin{tikzpicture}[
    x=1mm, y=0.9mm,
    every node/.style={font=\scriptsize},
    head/.style={draw, fill=blue!8, minimum width=20mm, minimum height=7mm, font=\footnotesize},
    attacker/.style={head, fill=red!10},
    life/.style={densely dashed, gray!55},
    frame/.style={draw=gray!60!black},
    tag/.style={font=\footnotesize\bfseries, anchor=north west, inner sep=2pt},
    call/.style={-{Stealth}, thick},
    reply/.style={-{Straight Barb}, thick, dashed},
    pay/.style={red!70!black},
    plain/.style={gray!60!black},
    lbl/.style={above, inner sep=1.5pt, text=black, font=\scriptsize}
]
    % lifelines
    \node[attacker] at ( 12, 0) {útočník};

    \node[head]     at (108, 0) {server aplikace};
    \node[head, minimum width=18mm] at (143, 0) {databáze};
    \begin{scope}[on background layer]
        \foreach \x in {12, 60, 108, 143} { \draw[life] (\x, -3.5) -- (\x, -146); }
    \end{scope}
    % frame with a pentagon tag
    \newcommand\sdframe[3]{%
        \draw[frame] (0, #1) rectangle (152, #1-#2);
        \draw[frame, fill=white] (0, #1) -- (#3, #1) -- (#3, #1-4.5) -- (#3-2, #1-6.5) -- (0, #1-6.5) -- cycle;
        \node[tag] at (0, #1) {sd};}
    % self message: a small loop on the lifeline at x, y with a label to the right
    \newcommand\selfmsg[4]{%
        \draw[call, #4] (#1, #2) -- (#1+6, #2) -- (#1+6, #2-5) -- (#1+0.6, #2-5);
        \node[anchor=west, inner sep=1.5pt] at (#1+7, #2-2.5) {#3};}

    \sdframe{-6}{38}{58}
    \node[tag, xshift=5mm] at (0, -6) {Reflektované XSS (typ~I)};

    \draw[call, pay] (108, -67) -- node[lbl] {2: uložení} (143, -67);
    \draw[call, plain] (60, -73) -- node[lbl] {3: později \texttt{GET /clanek}} (108, -73);

    \sdframe{-104}{41}{56}
    \node[tag, xshift=5mm] at (0, -104) {DOM-based XSS (typ~0)};
    \draw[call, pay] (12, -117) -- node[lbl] {1: odkaz \texttt{\#<img src=x onerror=…>}} (60, -117);
    \draw[call, plain] (60, -123) -- node[lbl] {2: \texttt{GET /} bez fragmentu \texttt{\#…}} (108, -123);
    \draw[reply, plain] (108, -129) -- node[lbl] {3: nezměněná stránka} (60, -129);

    \selfmsg{60}{-138}{5: spuštění skriptu}{pay}

    \draw[call, pay] (2, -151) -- (14, -151);
